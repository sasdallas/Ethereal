if(NOT CMAKE_SYSROOT)
    if("$ENV{SYSROOT}" STREQUAL "")
        message(FATAL_ERROR "Set SYSROOT or pass -DCMAKE_SYSROOT=/path/to/sysroot")
    endif()
    set(CMAKE_SYSROOT "$ENV{SYSROOT}" CACHE PATH "Ethereal sysroot")
endif()

set(CMAKE_SYSTEM_NAME Ethereal)
set(CMAKE_SYSTEM_PROCESSOR x86_64)

set(CMAKE_INSTALL_PREFIX "/usr" CACHE PATH "Install prefix")

set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE BOTH)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED TRUE)
set(CMAKE_CXX_COMPILER x86_64-ethereal-g++)
set(CMAKE_CXX_COMPILER_WORKS TRUE)

set(CMAKE_C_COMPILER x86_64-ethereal-gcc)
set(CMAKE_C_COMPILER_WORKS TRUE)

set(CMAKE_SHARED_LIBRARY_SONAME_C_FLAG "-Wl,-soname,")
set(CMAKE_SHARED_LIBRARY_SONAME_CXX_FLAG "-Wl,-soname,")

set(CMAKE_SHARED_LIBRARY_RUNTIME_C_FLAG "-Wl,-rpath,")
set(CMAKE_SHARED_LIBRARY_RUNTIME_CXX_FLAG "-Wl,-rpath,")

set(CMAKE_SHARED_LIBRARY_RUNTIME_C_FLAG_SEP ":")
set(CMAKE_SHARED_LIBRARY_RUNTIME_CXX_FLAG_SEP ":")
